cmake_minimum_required(VERSION 3.18)
project(quick_CFD_for_NNLCI_airfoil_opt_case LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release)
endif()
option(CFD_CUDA "Enable NVIDIA CUDA backend" ON)
option(CFD_OPENMP "Enable CPU OpenMP" OFF)
option(CFD_BUILD_3D "Build the explicit-face and procedural-AMR 3-D tools" ON)
if(CFD_CUDA)
 if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures; RTX4090 default" FORCE)
 endif()
 enable_language(CUDA)
 set(CMAKE_CUDA_STANDARD 17)
 find_package(CUDAToolkit REQUIRED)
endif()
add_library(cfd_core STATIC src/euler1d.cpp src/euler2d.cpp src/implicit2d.cpp src/ib2d.cpp src/cutcell2d.cpp src/surface2d.cpp src/static_amr2d.cpp src/static_amr2d_io.cpp src/static_amr2d_gpu_stub.cpp src/static_amr2d_binary.cpp src/static_amr2d_boundary.cpp src/static_amr2d_viscous.cpp src/static_amr2d_sa.cpp)
target_include_directories(cfd_core PUBLIC src)
if(CFD_OPENMP)
 find_package(OpenMP REQUIRED COMPONENTS CXX)
 target_link_libraries(cfd_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(CFD_CUDA)
 target_sources(cfd_core PRIVATE src/euler2d_cuda.cu src/static_amr2d_cuda.cu)
 target_compile_definitions(cfd_core PUBLIC CFD_HAVE_CUDA=1)
 target_link_libraries(cfd_core PUBLIC CUDA::cudart)
 set_target_properties(cfd_core PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
add_executable(quick_airfoil src/quick_airfoil.cpp)
target_link_libraries(quick_airfoil PRIVATE cfd_core)
# Frozen driver for reproducibility; not needed for the user pipeline.
add_executable(baseline_airfoil_suite EXCLUDE_FROM_ALL src/airfoil_suite.cpp)
target_link_libraries(baseline_airfoil_suite PRIVATE cfd_core)
if(CFD_BUILD_3D)
 add_subdirectory(solvers/3d)
endif()
